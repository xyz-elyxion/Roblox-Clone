  # Allow the user to enable/disable all system utility library options by
  # defining LINEARBUILD_USE_SYSTEM_LIBRARIES or LINEARBUILD_USE_SYSTEM_LIBRARY_${util}.
  set(UTILITIES BZIP2 CPPDAP CURL EXPAT FORM JSONCPP LIBARCHIVE LIBLZMA LIBRHASH LIBUV NGHTTP2 ZLIB ZSTD)
  foreach(util IN LISTS UTILITIES)
    if(NOT DEFINED LINEARBUILD_USE_SYSTEM_LIBRARY_${util}
        AND DEFINED LINEARBUILD_USE_SYSTEM_LIBRARIES)
      set(LINEARBUILD_USE_SYSTEM_LIBRARY_${util} "${LINEARBUILD_USE_SYSTEM_LIBRARIES}")
    endif()
    if(DEFINED LINEARBUILD_USE_SYSTEM_LIBRARY_${util})
      if(LINEARBUILD_USE_SYSTEM_LIBRARY_${util})
        set(LINEARBUILD_USE_SYSTEM_LIBRARY_${util} ON)
      else()
        set(LINEARBUILD_USE_SYSTEM_LIBRARY_${util} OFF)
      endif()
      if(LINEARBUILD_BOOTSTRAP)
        unset(LINEARBUILD_USE_SYSTEM_LIBRARY_${util} CACHE)
      endif()
      string(TOLOWER "${util}" lutil)
      set(LINEARBUILD_USE_SYSTEM_${util} "${LINEARBUILD_USE_SYSTEM_LIBRARY_${util}}"
        CACHE BOOL "Use system-installed ${lutil}" FORCE)
    elseif(util STREQUAL "CURL" AND APPLE)
      # macOS provides a curl with backends configured by Apple.
      set(LINEARBUILD_USE_SYSTEM_LIBRARY_${util} ON)
    else()
      set(LINEARBUILD_USE_SYSTEM_LIBRARY_${util} OFF)
    endif()
  endforeach()
  if(LINEARBUILD_BOOTSTRAP)
    unset(LINEARBUILD_USE_SYSTEM_LIBRARIES CACHE)
  endif()

#-----------------------------------------------------------------------
# a macro to determine the generator and ctest executable to use
# for testing. Simply to improve readability of the main script.
#-----------------------------------------------------------------------
macro(LINEARBUILD_SETUP_TESTING)
  if(BUILD_TESTING)
    set(LINEARBUILD_TEST_SYSTEM_LIBRARIES 0)
    foreach(util IN ITEMS CURL EXPAT ZLIB)
      if(LINEARBUILD_USE_SYSTEM_${util})
        set(LINEARBUILD_TEST_SYSTEM_LIBRARIES 1)
      endif()
    endforeach()

    # This variable is set by linearbuild, however to
    # test linearbuild we want to make sure that
    # the ctest from this linearbuild is used for testing
    # and not the ctest from the linearbuild building and testing
    # linearbuild.
    if(LinearBuild_TEST_EXTERNAL_LINEARBUILD)
      set(LINEARBUILD_CTEST_COMMAND "${LinearBuild_TEST_EXTERNAL_LINEARBUILD}/ctest")
      set(LINEARBUILD_LINEARBUILD_COMMAND "${LinearBuild_TEST_EXTERNAL_LINEARBUILD}/linearbuild")
      set(LINEARBUILD_CPACK_COMMAND "${LinearBuild_TEST_EXTERNAL_LINEARBUILD}/cpack")
      foreach(exe IN ITEMS linearbuild ctest cpack)
        add_executable(${exe} IMPORTED)
        set_property(TARGET ${exe} PROPERTY IMPORTED_LOCATION ${LinearBuild_TEST_EXTERNAL_LINEARBUILD}/${exe})
      endforeach()
    else()
      set(LINEARBUILD_CTEST_COMMAND "${LinearBuild_BIN_DIR}/ctest")
      set(LINEARBUILD_LINEARBUILD_COMMAND "${LinearBuild_BIN_DIR}/linearbuild")
      set(LINEARBUILD_CPACK_COMMAND "${LinearBuild_BIN_DIR}/cpack")
    endif()
  endif()

  # configure some files for testing. Tests/ is not carried by this tree, so the
  # NoDartCoverage marker is only configured when that directory exists.
  if(EXISTS ${LinearBuild_SOURCE_DIR}/Tests/.NoDartCoverage)
    configure_file(Tests/.NoDartCoverage Tests/.NoDartCoverage)
  endif()
  configure_file(CTestCustom.linearbuild.in CTestCustom.linearbuild @ONLY)
endmacro()

option(LinearBuild_RUN_CLANG_TIDY "Run clang-tidy with the compiler." OFF)
if(LinearBuild_RUN_CLANG_TIDY)
  if(LinearBuild_SOURCE_DIR STREQUAL LinearBuild_BINARY_DIR)
    message(FATAL_ERROR "LinearBuild_RUN_CLANG_TIDY requires an out-of-source build!")
  endif()
  find_program(CLANG_TIDY_COMMAND NAMES clang-tidy)
  if(NOT CLANG_TIDY_COMMAND)
    message(FATAL_ERROR "LinearBuild_RUN_CLANG_TIDY is ON but clang-tidy is not found!")
  endif()
  set(LINEARBUILD_CXX_CLANG_TIDY "${CLANG_TIDY_COMMAND}")

  option(LinearBuild_USE_CLANG_TIDY_MODULE "Use LinearBuild's clang-tidy module." OFF)
  if(LinearBuild_USE_CLANG_TIDY_MODULE)
    find_library(LinearBuild_CLANG_TIDY_MODULE NAMES linearbuild-clang-tidy-module DOC "Location of the clang-tidy module")
    if(NOT LinearBuild_CLANG_TIDY_MODULE)
      message(FATAL_ERROR "LinearBuild_USE_CLANG_TIDY_MODULE is ON but linearbuild-clang-tidy-module is not found!")
    endif()
    list(APPEND LINEARBUILD_CXX_CLANG_TIDY "--load=${LinearBuild_CLANG_TIDY_MODULE}")
    set_property(DIRECTORY APPEND PROPERTY LINEARBUILD_CONFIGURE_DEPENDS "${LinearBuild_CLANG_TIDY_MODULE}")
  endif()

  set(LinearBuild_CLANG_TIDY_EXPORT_FIXES_DIR "" CACHE PATH "Directory to put clang-tidy fix files in.")
  mark_as_advanced(LinearBuild_CLANG_TIDY_EXPORT_FIXES_DIR)
  if(LinearBuild_CLANG_TIDY_EXPORT_FIXES_DIR)
    if(NOT IS_ABSOLUTE "${LinearBuild_CLANG_TIDY_EXPORT_FIXES_DIR}")
      message(FATAL_ERROR "LinearBuild_CLANG_TIDY_EXPORT_FIXES_DIR must be an absolute path!")
    endif()
    set(LINEARBUILD_CXX_CLANG_TIDY_EXPORT_FIXES_DIR "${LinearBuild_CLANG_TIDY_EXPORT_FIXES_DIR}")
  endif()

  # Create a preprocessor definition that depends on .clang-tidy content so
  # the compile command will change when .clang-tidy changes.  This ensures
  # that a subsequent build re-runs clang-tidy on all sources even if they
  # do not otherwise need to be recompiled.  Nothing actually uses this
  # definition.  We add it to targets on which we run clang-tidy just to
  # get the build dependency on the .clang-tidy file.
  file(SHA1 ${LINEARBUILD_CURRENT_SOURCE_DIR}/.clang-tidy clang_tidy_sha1)
  set(CLANG_TIDY_DEFINITIONS "CLANG_TIDY_SHA1=${clang_tidy_sha1}")
  unset(clang_tidy_sha1)
  if(LinearBuild_USE_CLANG_TIDY_MODULE)
    file(SHA1 "${LinearBuild_CLANG_TIDY_MODULE}" clang_tidy_module_sha1)
    list(APPEND CLANG_TIDY_DEFINITIONS "CLANG_TIDY_MODULE_SHA1=${clang_tidy_module_sha1}")
    unset(clang_tidy_module_sha1)
  endif()

  configure_file(.clang-tidy .clang-tidy COPYONLY)
endif()

        NOT "x${LINEARBUILD_C_SIMULATE_ID}" STREQUAL "xMSVC") OR
       LINEARBUILD_C_COMPILER_ID STREQUAL "AppleClang" OR
       LINEARBUILD_C_COMPILER_ID STREQUAL "LCC")
      set(C_FLAGS_LIST -Wcast-align -Wchar-subscripts
                       -Wall -W -Wpointer-arith -Wwrite-strings -Wformat-security
                       -Wmissing-format-attribute -fno-common -Wundef
                       -Werror=implicit-function-declaration
                       -Wstrict-prototypes
      )
      set(CXX_FLAGS_LIST -Wnon-virtual-dtor -Wcast-align -Wchar-subscripts -Wall -W
                         -Wshadow -Wpointer-arith -Wformat-security -Wundef
      )
      if(LINEARBUILD_CXX_COMPILER_ID MATCHES "Clang" AND
          NOT (LINEARBUILD_CXX_COMPILER_ID STREQUAL "AppleClang" AND LINEARBUILD_CXX_COMPILER_VERSION VERSION_LESS 11))
        list(APPEND CXX_FLAGS_LIST
          -Wundefined-func-template
        )
      endif()

      foreach(FLAG_LANG IN ITEMS C CXX)
        foreach(FLAG IN LISTS ${FLAG_LANG}_FLAGS_LIST)
          if(NOT " ${LINEARBUILD_${FLAG_LANG}_FLAGS} " MATCHES " ${FLAG} ")
            string(APPEND LINEARBUILD_${FLAG_LANG}_FLAGS " ${FLAG}")
          endif()
        endforeach()
      endforeach()

      unset(C_FLAGS_LIST)
      unset(CXX_FLAGS_LIST)
    endif()
  endif()

  # build the remaining subdirectories
  add_subdirectory(Source)
  add_subdirectory(Utilities)
endif()
