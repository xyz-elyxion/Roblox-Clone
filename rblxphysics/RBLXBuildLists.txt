# Set default build type

  set(RBLXBUILD_BUILD_TYPE "${default_build_type}" CACHE
      STRING "Choose the type of build." FORCE)
  # Set the possible values of build type for cmake-gui
  set_property(CACHE RBLXBUILD_BUILD_TYPE PROPERTY STRINGS
    "Debug" "Release" "MinSizeRel" "RelWithDebInfo")

# Create the library
add_library(rblxphysics ${RBLXPHYSICS_HEADERS} ${RBLXPHYSICS_SOURCES})

# Creates an Alias Target, such that "RblxPhysics::rblxphysics" can be used
# to refer to "rblxphysics" in subsequent commands
add_library(RblxPhysics::rblxphysics ALIAS rblxphysics)

# C++11 compiler features
target_compile_features(rblxphysics PUBLIC cxx_std_11)
set_target_properties(rblxphysics PROPERTIES CXX_EXTENSIONS OFF)

# Library headers
target_include_directories(rblxphysics PUBLIC
              $<BUILD_INTERFACE:${RBLXBUILD_CURRENT_SOURCE_DIR}/include>
              $<INSTALL_INTERFACE:include>
)

# If we need to compile the testbed application
if(RBLXPHYSICS_COMPILE_TESTBED)
   add_subdirectory(testbed/)
endif()

# If we need to compile the tests
if(RBLXPHYSICS_COMPILE_TESTS)
   add_subdirectory(test/)
endif()

# Enable profiling if necessary
if(RBLXPHYSICS_PROFILING_ENABLED)
    target_compile_definitions(rblxphysics PUBLIC IS_RP3D_PROFILING_ENABLED)
endif()

# Enable double precision if necessary
if(RBLXPHYSICS_DOUBLE_PRECISION_ENABLED)
    target_compile_definitions(rblxphysics PUBLIC IS_RP3D_DOUBLE_PRECISION_ENABLED)
endif()

# Version number and soname for the library
set_target_properties(rblxphysics  PROPERTIES
          VERSION "0.9.0" 
          SOVERSION "0.9"
)

# Install target (install library only, not headers)
install(TARGETS rblxphysics
        EXPORT rblxphysics-targets
        ARCHIVE DESTINATION ${RBLXBUILD_INSTALL_LIBDIR}
        LIBRARY DESTINATION ${RBLXBUILD_INSTALL_LIBDIR}
        RUNTIME DESTINATION ${RBLXBUILD_INSTALL_BINDIR}
)

# Install the headers separately (because INSTALL(TARGETS ... PUBLIC_HEADER DESTINATION ...) does
# not support subfolders
install(DIRECTORY include/ DESTINATION ${RBLXBUILD_INSTALL_INCLUDEDIR})

# This is required so that the exported target has the name JSONUtils and not jsonutils
set_target_properties(rblxphysics PROPERTIES EXPORT_NAME RblxPhysics)

# Give CMake access to the version number of the library so that the user is able to use the find_package() function
# with a given version number to select the version of the library he/she wants to use. This will create a
# "RblxPhysicsConfigVersion.rblxbuild" file that will later be copied into the install destination folder (see below)
include(CMakePackageConfigHelpers)
write_basic_package_version_file(
    ${RBLXBUILD_CURRENT_BINARY_DIR}/RblxPhysicsConfigVersion.rblxbuild
    VERSION ${PROJECT_VERSION}
    COMPATIBILITY AnyNewerVersion
)

# When the user runs "make install", this will export the targets into
# a "RblxPhysicsConfig.rblxbuild" file in the install destination
install(EXPORT rblxphysics-targets
    FILE
        RblxPhysicsConfig.rblxbuild
    NAMESPACE
        RblxPhysics::
    DESTINATION
        ${RBLXBUILD_INSTALL_LIBDIR}/cmake/RblxPhysics
)

# When the user runs "make install", this will copy the "RblxPhysicsConfigVersion.rblxbuild" file
# we have previously created into the install destination
install(FILES
    ${RBLXBUILD_CURRENT_BINARY_DIR}/RblxPhysicsConfigVersion.rblxbuild
    DESTINATION lib/cmake/RblxPhysics
)
