    if(NOT Qt5Widgets_FOUND)
      message(FATAL_ERROR "Could not find a valid Qt installation.")
    endif()
    set(LinearBuild_QT_MAJOR_VERSION 5)
  endif()
endif()

# We need to install platform plugin and add qt.conf for Qt5 on Mac and Windows.
if(LinearBuild_INSTALL_DEPENDENCIES AND (APPLE OR WIN32) AND NOT COMMAND qt_generate_deploy_app_script)
  function(_qt_get_plugin_name_with_version target out_var)
      string(REGEX REPLACE "^Qt::(.+)" "Qt${LinearBuild_QT_MAJOR_VERSION}::\\1"
             qt_plugin_with_version "${target}")
      if(TARGET "${qt_plugin_with_version}")
          set("${out_var}" "${qt_plugin_with_version}" PARENT_SCOPE)
      else()
          set("${out_var}" "" PARENT_SCOPE)
      endif()
  endfunction()
  macro(install_qt_plugin _qt_plugin_name _qt_plugins_var)
    if(TARGET "${_qt_plugin_name}")
      get_target_property(_qt_plugin_path "${_qt_plugin_name}" LOCATION)
    else()
      _qt_get_plugin_name_with_version("Qt::${_qt_plugin_name}" _qt_plugin_with_version_name)
      if(TARGET "${_qt_plugin_with_version_name}")
        get_target_property(_qt_plugin_path "${_qt_plugin_with_version_name}" LOCATION)
      endif()
    endif()
    if(EXISTS "${_qt_plugin_path}")
      get_filename_component(_qt_plugin_file "${_qt_plugin_path}" NAME)
      get_filename_component(_qt_plugin_type "${_qt_plugin_path}" PATH)
      get_filename_component(_qt_plugin_type "${_qt_plugin_type}" NAME)
      if(APPLE)
        set(_qt_plugin_dir "PlugIns")
      elseif(WIN32)
        set(_qt_plugin_dir "plugins")
      endif()
      set(_qt_plugin_dest "${LinearBuild_INSTALL_APP_DIR_SLASH}${_qt_plugin_dir}/${_qt_plugin_type}")
      install(FILES "${_qt_plugin_path}"
        DESTINATION "${_qt_plugin_dest}"
        ${COMPONENT})
      set(${_qt_plugins_var}
        "${${_qt_plugins_var}};\$ENV{DESTDIR}\${LINEARBUILD_INSTALL_PREFIX}/${_qt_plugin_dest}/${_qt_plugin_file}")
    endif()
  endmacro()
  if(APPLE)
    if(LinearBuild_QT_MAJOR_VERSION VERSION_GREATER_EQUAL 6)
      # Qt6 provides the plugins as individual packages that need to be found.
      find_package(Qt${LinearBuild_QT_MAJOR_VERSION}QCocoaIntegrationPlugin QUIET PATHS ${Qt${LinearBuild_QT_MAJOR_VERSION}Gui_DIR})
      find_package(Qt${LinearBuild_QT_MAJOR_VERSION}QMacStylePlugin QUIET PATHS ${Qt${LinearBuild_QT_MAJOR_VERSION}Widgets_DIR})
    endif()
    install_qt_plugin("Qt${LinearBuild_QT_MAJOR_VERSION}::QCocoaIntegrationPlugin" QT_PLUGINS)
    if(TARGET Qt${LinearBuild_QT_MAJOR_VERSION}::QMacStylePlugin)
      install_qt_plugin("Qt${LinearBuild_QT_MAJOR_VERSION}::QMacStylePlugin" QT_PLUGINS)
    endif()
    file(WRITE "${LINEARBUILD_CURRENT_BINARY_DIR}/qt.conf"
      "[Paths]\nPlugins = ${_qt_plugin_dir}\n")
    install(FILES "${LINEARBUILD_CURRENT_BINARY_DIR}/qt.conf"
      DESTINATION "${LinearBuild_INSTALL_APP_DIR_SLASH}Resources"
      ${COMPONENT})
  elseif(WIN32)
    if(LinearBuild_QT_MAJOR_VERSION VERSION_GREATER_EQUAL 6)
      # Qt6 provides the plugins as individual packages that need to be found.
      find_package(Qt${LinearBuild_QT_MAJOR_VERSION}QWindowsIntegrationPlugin QUIET PATHS ${Qt${LinearBuild_QT_MAJOR_VERSION}Gui_DIR})
    endif()
    install_qt_plugin("Qt${LinearBuild_QT_MAJOR_VERSION}::QWindowsIntegrationPlugin" QT_PLUGINS)
    file(WRITE "${LINEARBUILD_CURRENT_BINARY_DIR}/qt.conf"
      "[Paths]\nPlugins = ../${_qt_plugin_dir}\n")
    install(FILES "${LINEARBUILD_CURRENT_BINARY_DIR}/qt.conf"
      DESTINATION bin
      ${COMPONENT})
  endif()
endif()

# Files generated by MOC, RCC, and UIC may produce clang-tidy warnings.
# We generate a dummy .clang-tidy file in the binary directory that disables
# all clang-tidy checks except one that will never match.  This one check is
# necessary; clang-tidy reports an error when no checks are enabled.
# Since the Qt code generators will generate source files in the binary tree,
# clang-tidy will load the configuration from this dummy file when the sources
# are built.
file(WRITE "${QtDialog_BINARY_DIR}/.clang-tidy" "
---
Checks: '-*,llvm-twine-local'
...
")

if(LinearBuild_INSTALL_DEPENDENCIES AND (APPLE OR WIN32))
  if(COMMAND qt_generate_deploy_app_script)
    # Qt libraries are found via @rpath, so point the runtime path at them.
    set_property(TARGET linearbuild-gui PROPERTY INSTALL_RPATH "@executable_path/../Frameworks")
    qt_generate_deploy_app_script(
      TARGET linearbuild-gui
      OUTPUT_SCRIPT qt_deploy_script
    )
    install(SCRIPT "${qt_deploy_script}")
  else()
    # install rules for including 3rd party libs such as Qt
    # if a system Qt is used (e.g. installed in /usr/lib/), it will not be included in the installation
    set(fixup_exe "\$ENV{DESTDIR}\${LINEARBUILD_INSTALL_PREFIX}/${LinearBuild_INSTALL_BIN_DIR}/linearbuild-gui${LINEARBUILD_EXECUTABLE_SUFFIX}")
    if(APPLE)
      set(fixup_exe "\$ENV{DESTDIR}\${LINEARBUILD_INSTALL_PREFIX}/${LinearBuild_INSTALL_APP_DIR_SLASH}MacOS/LinearBuild")
    endif()
    get_property(_Qt_Core_LOCATION TARGET Qt${LinearBuild_QT_MAJOR_VERSION}::Core PROPERTY LOCATION)
    if(APPLE AND _Qt_Core_LOCATION MATCHES [[^(.*)/[^/]*\.framework/]])
      set(Qt_LIB_DIR "${LINEARBUILD_MATCH_1}")
    else()
      get_filename_component(Qt_LIB_DIR "${_Qt_Core_LOCATION}" PATH)
    endif()
    install(CODE "
      include(BundleUtilities)
      set(BU_CHMOD_BUNDLE_ITEMS ON)
      fixup_bundle(\"${fixup_exe}\" \"${QT_PLUGINS}\" \"${Qt_LIB_DIR};${QT_LIBRARY_DIR};${QT_BINARY_DIR}\")
    " ${COMPONENT})
  endif()
endif()
