# rblxphysics's testbed, unit tests and install rules are off by default, so this adds
# only the rblxphysics library.
add_subdirectory(${RBLXBUILD_CURRENT_SOURCE_DIR}/rblxphysics)

# rblxphysics 0.9.0 depends on headers it does not include itself: DefaultLogger.h uses
# std::chrono::system_clock without <chrono>, and the public headers use the
# fixed-width integer types without <cstdint>. Older toolchains saw both through
# other standard headers, current ones do not, so force them in for everything
# compiled against the library.
if(TARGET rblxphysics)
    if(MSVC)
        target_compile_options(rblxphysics PUBLIC
            $<$<COMPILE_LANGUAGE:CXX>:/FIchrono>
        )
    else()
        target_compile_options(rblxphysics PUBLIC
            $<$<COMPILE_LANGUAGE:CXX>:-include>
            $<$<COMPILE_LANGUAGE:CXX>:cstdint>
        )
    endif()
endif()

function(configure_rblx_target target_name)
    target_include_directories(${target_name} PRIVATE
        ${RBLXBUILD_CURRENT_SOURCE_DIR}/include
        ${rblxstb_SOURCE_DIR}
        ${rblxui_SOURCE_DIR}
        ${rblxui_SOURCE_DIR}/backends
        ${rblxgizmo_SOURCE_DIR}
        ${rblxgizmo_SOURCE_DIR}/src
        ${rblxgl_SOURCE_DIR}/include
    )

    if(RBLX_ENABLE_WARNINGS)
        if(MSVC)
            target_compile_options(${target_name} PRIVATE /W4 /permissive-)
        else()
            target_compile_options(${target_name} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endif()
endfunction()

if(RBLX_BUILD_UPDATER)
    # The installer and the auto updater are one target. It reads manifest.json,
    # downloads only the files whose SHA-256 differs, replaces them through a
    # rename, and starts the game - which is what the installed Play shortcut
    # runs. RBLXSetup (below) is the installer that puts a packaged folder on a
    # machine in the first place.
    #
    # Both show their work in a window (InstallUi), which is why they carry the
    # same window and RblxUI sources the editors do. The console commands are not a
    # fallback for a missing window: they are how a script and the packaging step
    # drive these tools, and they keep working exactly as before.
    add_executable(RBLXUpdater
        src/UpdaterMain.cpp
        src/Updater.cpp
        src/WebJson.cpp
        src/WebCrypto.cpp
        src/HttpClient.cpp
        src/InstallUi.cpp
        ${RBLXUI_SOURCES}
    )
    configure_rblx_target(RBLXUpdater)
    target_link_libraries(RBLXUpdater PRIVATE rblxwin rblxgl)
    target_link_libraries(RBLXUpdater PRIVATE ws2_32)

    # The setup is the installer: it copies the packaged folder into a per-user
    # location, records what it put there as manifest.json, adds the shortcuts and
    # the Add/Remove Programs entry, and is its own uninstaller. It never touches
    # the network - the payload is the folder it was run from - so it links the
    # file and hash helpers the updater uses, the window, and nothing else.
    add_executable(RBLXSetup
        src/SetupMain.cpp
        src/Updater.cpp
        src/WebJson.cpp
        src/WebCrypto.cpp
        src/InstallUi.cpp
        ${RBLXUI_SOURCES}
    )
    configure_rblx_target(RBLXSetup)
    target_link_libraries(RBLXSetup PRIVATE rblxwin rblxgl)
    # IShellLink and IPersistFile are COM, the folder lookups are shell32, and
    # the uninstall entry is written with the registry API.
    target_link_libraries(RBLXSetup PRIVATE ole32 shell32 advapi32 uuid)
endif()

        # website uses, so the client links it rather than a second copy.
        src/HardwareId.cpp
        src/HttpClient.cpp
        # The client mixes its own sounds; the bank is synthesised, not shipped.
        src/Audio.cpp
        src/SoundBank.cpp
        ${SHARED_SOURCES}
        ${RBLXUI_SOURCES}
    )
    configure_rblx_target(Client)
    target_link_libraries(Client PRIVATE rblxwin rblxmath rblxphysics rblxgl)
    # winmm is the OS mixer waveOut lives in; no other audio dependency exists.
    target_link_libraries(Client PRIVATE ws2_32 shell32 comdlg32 winmm)
    if(TARGET RBLXAntiCheat)
        target_compile_definitions(Client PRIVATE RBLX_ANTICHEAT)
        target_link_libraries(Client PRIVATE RBLXAntiCheat)
    endif()
    if(MSVC)
        # Hardening for the executable players actually run. Control-flow guard
        # rejects indirect calls that do not land on a valid target, so a patched
        # call site in the anti-cheat path cannot simply be redirected, and full
        # ASLR keeps the image from loading at a predictable base. Identical-code
        # folding makes the individual checks harder to tell apart when reading
        # the disassembly.
        target_compile_options(Client PRIVATE
            $<$<CONFIG:Release>:/guard:cf>
            $<$<CONFIG:Release>:/Gy>
            $<$<CONFIG:Release>:/GS>
        )
        target_link_options(Client PRIVATE
            $<$<CONFIG:Release>:/guard:cf>
            $<$<CONFIG:Release>:/DYNAMICBASE>
            $<$<CONFIG:Release>:/HIGHENTROPYVA>
            $<$<CONFIG:Release>:/OPT:REF>
            $<$<CONFIG:Release>:/OPT:ICF>
        )
    endif()
endif()
