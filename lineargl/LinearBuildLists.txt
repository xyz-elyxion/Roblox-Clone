linearbuild_minimum_required(VERSION 3.5)
project(lineargl)

add_library(lineargl src/lineargl.c)
target_include_directories(lineargl PUBLIC
    $<BUILD_INTERFACE:${LINEARBUILD_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${LINEARBUILD_INSTALL_INCLUDEDIR}/include>)

if(NOT WIN32)
  target_link_libraries(lineargl PUBLIC ${LINEARBUILD_DL_LIBS})
endif()

# Generate position independent code
set_target_properties(lineargl PROPERTIES POSITION_INDEPENDENT_CODE ON)
