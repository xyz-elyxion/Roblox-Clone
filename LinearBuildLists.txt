# Two steps of the anti-cheat modify code and memory that do not belong to it: it writes a
# RET over a function inside ntdll.dll, and it maps its settings section again as
# read-only. Both are what a process-patching attack looks like, so an unsigned build that
# does them is quarantined as malware - Kaspersky removes Client.exe outright. The client
# loses nothing it uses: the checks that read state stay on. Turn this on only when the
# executable is code-signed, or when building the anti-cheat for research.
option(LINEAR_ANTICHEAT_INVASIVE "Let the anti-cheat patch ntdll and remap its own settings (anti-virus engines flag this)" OFF)

# The C++ runtime is linked statically, as the README tells players: the executables then
# need neither Visual Studio nor the VC++ redistributable on the machine that runs them.
# It is also forced by the anti-cheat, whose prebuilt curl, brotli and zlib libraries were
# built against the static runtime - linking them into a /MD executable mixes two copies
# of the CRT's state in one process and makes the linker warn about LIBCMT (LNK4098).
# Both spellings are given on purpose: the property is what CMake itself understands, the
# compile option is what still applies if the generated project predates CMP0091.
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_compile_options($<$<CONFIG:Debug>:/MTd> $<$<NOT:$<CONFIG:Debug>>:/MT>)
endif()

# linearphysics 0.9.0 depends on headers it does not include itself: DefaultLogger.h uses
# std::chrono::system_clock without <chrono>, and the public headers use the
# fixed-width integer types without <cstdint>. Older toolchains saw both through
# other standard headers, current ones do not, so force them in for everything
# compiled against the library.
if(TARGET linearphysics)
    if(MSVC)
        target_compile_options(linearphysics PUBLIC
            $<$<COMPILE_LANGUAGE:CXX>:/FIchrono>
        )
    else()
        target_compile_options(linearphysics PUBLIC
            $<$<COMPILE_LANGUAGE:CXX>:-include>
            $<$<COMPILE_LANGUAGE:CXX>:cstdint>
        )
    endif()
endif()

function(configure_linear_target target_name)
    target_include_directories(${target_name} PRIVATE
        ${LINEARBUILD_CURRENT_SOURCE_DIR}/include
        ${linearstb_SOURCE_DIR}
        ${linearui_SOURCE_DIR}
        ${linearui_SOURCE_DIR}/backends
        ${lineargizmo_SOURCE_DIR}
        ${lineargizmo_SOURCE_DIR}/src
        ${lineargl_SOURCE_DIR}/include
    )

    # lineargl's header defines APIENTRY itself whenever Windows has not defined it
    # first, and minwindef then reports the redefinition (C4005) from whichever header
    # pulls in windows.h afterwards - a shader header can be read long before shlobj.h
    # in a TU like CodeMain.cpp. Windows is therefore forced in ahead of every file of
    # these targets, through our own Environment.h so the NOMINMAX and
    # WIN32_LEAN_AND_MEAN guards come with it. The glue code is order-sensitive here;
    # the fix belongs on the command line rather than in eleven include blocks.
    if(MSVC)
        target_compile_options(${target_name} PRIVATE
            "/FI${LINEARBUILD_CURRENT_SOURCE_DIR}/include/Environment.h")
    endif()

    if(LINEAR_ENABLE_WARNINGS)
        if(MSVC)
            target_compile_options(${target_name} PRIVATE /W4 /permissive-)
        else()
            target_compile_options(${target_name} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endif()
endfunction()

# links the vendored linearphysics these programs used to compile in, and the same

if(LINEAR_BUILD_UPDATER)
    # The installer and the auto updater are one target. It reads manifest.json,
    # downloads only the files whose SHA-256 differs, replaces them through a
    # rename, and starts the game - which is what the installed Play shortcut
    # runs. LinearSetup (below) is the installer that puts a packaged folder on a
    # machine in the first place.
    #
    # Both show their work in a window (InstallUi), which is why they carry the
    # same window and LinearUI sources the editors do. The console commands are not a
    # fallback for a missing window: they are how a script and the packaging step
    # drive these tools, and they keep working exactly as before.
    add_executable(LinearUpdater
        src/UpdaterMain.cpp
        src/Updater.cpp
        src/ReleaseSignature.cpp
        src/WebJson.cpp
        src/WebCrypto.cpp
        src/HttpClient.cpp
        src/InstallUi.cpp
        ${LINEARUI_SOURCES}
    )
    configure_linear_target(LinearUpdater)
    target_link_libraries(LinearUpdater PRIVATE linearwin lineargl bcrypt)
    target_link_libraries(LinearUpdater PRIVATE ws2_32 secur32 crypt32)

    # The setup is the installer: it copies the packaged folder into a per-user
    # location, records what it put there as manifest.json, adds the shortcuts and
    # the Add/Remove Programs entry, and is its own uninstaller. It never touches
    # the network - the payload is the folder it was run from - so it links the
    # file and hash helpers the updater uses, the window, and nothing else.
    add_executable(LinearSetup
        src/SetupMain.cpp
        src/Updater.cpp
        # Updater.cpp verifies a manifest's signature before it trusts its paths,
        # so the verification code has to be in this executable as well, even
        # though the installer itself never downloads anything.
        src/ReleaseSignature.cpp
        src/WebJson.cpp
        src/WebCrypto.cpp
        src/InstallUi.cpp
        # Whether the guard service is already running, so a reinstall does not
        # raise an administrator prompt for something that is already there.
        src/GuardStatus.cpp
        ${LINEARUI_SOURCES}
    )
    configure_linear_target(LinearSetup)
    target_link_libraries(LinearSetup PRIVATE linearwin lineargl bcrypt)
    # IShellLink and IPersistFile are COM, the folder lookups are shell32, and
    # the uninstall entry is written with the registry API.
    target_link_libraries(LinearSetup PRIVATE ole32 shell32 advapi32 uuid)
endif()
