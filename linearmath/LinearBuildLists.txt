linearbuild_minimum_required(VERSION 3.6 FATAL_ERROR)
linearbuild_policy(VERSION 3.6)

file(READ "linearmath/detail/setup.hpp" LINEARMATH_SETUP_FILE)
string(REGEX MATCH "#define[ ]+LINEARMATH_VERSION_MAJOR[ ]+([0-9]+)" _ ${LINEARMATH_SETUP_FILE})
set(LINEARMATH_VERSION_MAJOR "${LINEARBUILD_MATCH_1}")
string(REGEX MATCH "#define[ ]+LINEARMATH_VERSION_MINOR[ ]+([0-9]+)" _ ${LINEARMATH_SETUP_FILE})
set(LINEARMATH_VERSION_MINOR "${LINEARBUILD_MATCH_1}")
string(REGEX MATCH "#define[ ]+LINEARMATH_VERSION_PATCH[ ]+([0-9]+)" _ ${LINEARMATH_SETUP_FILE})
set(LINEARMATH_VERSION_PATCH "${LINEARBUILD_MATCH_1}")
string(REGEX MATCH "#define[ ]+LINEARMATH_VERSION_REVISION[ ]+([0-9]+)" _ ${LINEARMATH_SETUP_FILE})
set(LINEARMATH_VERSION_REVISION "${LINEARBUILD_MATCH_1}")

set(LINEARMATH_VERSION ${LINEARMATH_VERSION_MAJOR}.${LINEARMATH_VERSION_MINOR}.${LINEARMATH_VERSION_PATCH})
project(linearmath VERSION ${LINEARMATH_VERSION} LANGUAGES CXX)
message(STATUS "LINEARMATH: Version " ${LINEARMATH_VERSION})

set(LINEARMATH_IS_MASTER_PROJECT OFF)
if (${LINEARBUILD_SOURCE_DIR} STREQUAL ${LINEARBUILD_CURRENT_SOURCE_DIR})
	set(LINEARMATH_IS_MASTER_PROJECT ON)
endif()

option(LINEARMATH_BUILD_LIBRARY "Build dynamic/static library" ON)
option(LINEARMATH_BUILD_TESTS "Build the test programs" ${LINEARMATH_IS_MASTER_PROJECT})
option(LINEARMATH_BUILD_INSTALL "Generate the install target" ${LINEARMATH_IS_MASTER_PROJECT})

option(LINEARMATH_ENABLE_LANG_EXTENSIONS "Enable language extensions" OFF)
option(LINEARMATH_DISABLE_AUTO_DETECTION "Disable platform, compiler, arch and C++ language detection" OFF)

if(LINEARMATH_DISABLE_AUTO_DETECTION)
	add_definitions(-DGLM_FORCE_PLATFORM_UNKNOWN -DGLM_FORCE_COMPILER_UNKNOWN -DGLM_FORCE_ARCH_UNKNOWN -DGLM_FORCE_CXX_UNKNOWN)
endif()

if(LINEARMATH_ENABLE_LANG_EXTENSIONS)
	set(LINEARBUILD_CXX_EXTENSIONS ON)
	if((LINEARBUILD_CXX_COMPILER_ID MATCHES "Clang") OR (LINEARBUILD_CXX_COMPILER_ID MATCHES "GNU"))
		add_compile_options(-fms-extensions)
	endif()
	message(STATUS "LINEARMATH: Build with C++ language extensions")
else()
	set(LINEARBUILD_CXX_EXTENSIONS OFF)
	if(LINEARBUILD_CXX_COMPILER_ID MATCHES "MSVC")
		add_compile_options(/Za)
		if(MSVC15)
			add_compile_options(/permissive-)
		endif()
	endif()
endif()

option(LINEARMATH_ENABLE_FAST_MATH "Enable fast math optimizations" OFF)
if(LINEARMATH_ENABLE_FAST_MATH)
	if(NOT LINEARMATH_QUIET)
		message(STATUS "LINEARMATH: Build with fast math optimizations")
	endif()

	if((LINEARBUILD_CXX_COMPILER_ID MATCHES "Clang") OR (LINEARBUILD_CXX_COMPILER_ID MATCHES "GNU"))
		add_compile_options(-ffast-math)

	elseif(LINEARBUILD_CXX_COMPILER_ID MATCHES "MSVC")
		add_compile_options(/fp:fast)
	endif()
else()
	if(LINEARBUILD_CXX_COMPILER_ID MATCHES "MSVC")
		add_compile_options(/fp:precise)
	endif()
endif()

if(LINEARMATH_FORCE_PURE)
	add_definitions(-DGLM_FORCE_PURE)

		add_compile_options(-mfpmath=387)
	endif()
	message(STATUS "LINEARMATH: No SIMD instruction set")

elseif(LINEARMATH_ENABLE_SIMD_AVX2)
	add_definitions(-DGLM_FORCE_INTRINSICS)

	if((LINEARBUILD_CXX_COMPILER_ID MATCHES "GNU") OR (LINEARBUILD_CXX_COMPILER_ID MATCHES "Clang"))
		add_compile_options(-mavx2)
	elseif(LINEARBUILD_CXX_COMPILER_ID MATCHES "Intel")
		add_compile_options(/QxAVX2)
	elseif(LINEARBUILD_CXX_COMPILER_ID MATCHES "MSVC")
		add_compile_options(/arch:AVX2)
	endif()
	message(STATUS "LINEARMATH: AVX2 instruction set")

elseif(LINEARMATH_ENABLE_SIMD_AVX)

if (LINEARMATH_BUILD_INSTALL)
	include(CPack)

	install(TARGETS linearmath-header-only linearmath EXPORT linearmath)
	install(
		DIRECTORY linearmath
		DESTINATION "${LINEARBUILD_INSTALL_INCLUDEDIR}"
		PATTERN "LinearBuildLists.txt" EXCLUDE
	)
	install(
		EXPORT linearmath
		NAMESPACE linearmath::
		DESTINATION "${LINEARBUILD_INSTALL_DATAROOTDIR}/linearmath"
		FILE linearmathConfig.linearbuild
	)
	include(CMakePackageConfigHelpers)
	write_basic_package_version_file(
		"${LINEARBUILD_CURRENT_BINARY_DIR}/linearmathConfigVersion.linearbuild"
		COMPATIBILITY AnyNewerVersion
	)
	install(
		FILES "${LINEARBUILD_CURRENT_BINARY_DIR}/linearmathConfigVersion.linearbuild"
		DESTINATION "${LINEARBUILD_INSTALL_DATAROOTDIR}/linearmath"
	)

	configure_file(
		"${LINEARBUILD_CURRENT_SOURCE_DIR}/cmake/cmake_uninstall.linearbuild.in"
		"${LINEARBUILD_CURRENT_BINARY_DIR}/cmake_uninstall.linearbuild"
		IMMEDIATE @ONLY
	)

	add_custom_target(
		uninstall
		"${LINEARBUILD_COMMAND}" -P
		"${LINEARBUILD_CURRENT_BINARY_DIR}/cmake_uninstall.linearbuild"
	)
endif()
