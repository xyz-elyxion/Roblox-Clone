# Disable warnings to avoid changing 3rd party code.
if(RBLXBUILD_C_COMPILER_ID MATCHES
    "^(GNU|LCC|Clang|AppleClang|IBMClang|XLClang|XL|VisualAge|SunPro|HP|Intel|IntelLLVM|NVHPC)$")
  set(RBLXBUILD_C_FLAGS "${RBLXBUILD_C_FLAGS} -w")
elseif(RBLXBUILD_C_COMPILER_ID STREQUAL "PathScale")
  set(RBLXBUILD_C_FLAGS "${RBLXBUILD_C_FLAGS} -woffall")
endif()

# Re-use some check result cache entries from cmcurl:
# * HAVE_ARPA_INET_H    (referenced in rblxbuildconfig.h.in)
# * HAVE_NETINET_IN_H   (referenced in rblxbuildconfig.h.in)
# * HAVE_SIZEOF_SSIZE_T (referenced here)
if(NOT HAVE_SIZEOF_SSIZE_T)
  set(ssize_t KWIML_INT_intptr_t)
endif()
configure_file(rblxbuildconfig.h.in config.h)

  )
