  set(UTILITIES BZIP2 CPPDAP CURL EXPAT FORM JSONCPP LIBARCHIVE LIBLZMA LIBRHASH LIBUV NGHTTP2 ZLIB ZSTD)
  foreach(util IN LISTS UTILITIES)
    if(NOT DEFINED RBLXBUILD_USE_SYSTEM_LIBRARY_${util}
        AND DEFINED RBLXBUILD_USE_SYSTEM_LIBRARIES)
      set(RBLXBUILD_USE_SYSTEM_LIBRARY_${util} "${RBLXBUILD_USE_SYSTEM_LIBRARIES}")
    endif()
    if(DEFINED RBLXBUILD_USE_SYSTEM_LIBRARY_${util})
      if(RBLXBUILD_USE_SYSTEM_LIBRARY_${util})
        set(RBLXBUILD_USE_SYSTEM_LIBRARY_${util} ON)
      else()
        set(RBLXBUILD_USE_SYSTEM_LIBRARY_${util} OFF)
      endif()
      if(RBLXBUILD_BOOTSTRAP)
        unset(RBLXBUILD_USE_SYSTEM_LIBRARY_${util} CACHE)
      endif()
      string(TOLOWER "${util}" lutil)
      set(RBLXBUILD_USE_SYSTEM_${util} "${RBLXBUILD_USE_SYSTEM_LIBRARY_${util}}"
        CACHE BOOL "Use system-installed ${lutil}" FORCE)
    elseif(util STREQUAL "CURL" AND APPLE)
      # macOS provides a curl with backends configured by Apple.
      set(RBLXBUILD_USE_SYSTEM_LIBRARY_${util} ON)
    else()
      set(RBLXBUILD_USE_SYSTEM_LIBRARY_${util} OFF)
    endif()
  endforeach()

    unset(RBLXBUILD_USE_SYSTEM_LIBRARIES CACHE)
  endif()

#-----------------------------------------------------------------------
# a macro to determine the generator and ctest executable to use
# for testing. Simply to improve readability of the main script.
#-----------------------------------------------------------------------
macro(RBLXBUILD_SETUP_TESTING)
  if(BUILD_TESTING)
    set(RBLXBUILD_TEST_SYSTEM_LIBRARIES 0)
    foreach(util IN ITEMS CURL EXPAT ZLIB)
      if(RBLXBUILD_USE_SYSTEM_${util})
        set(RBLXBUILD_TEST_SYSTEM_LIBRARIES 1)
      endif()
    endforeach()

    # This variable is set by rblxbuild, however to
    # test rblxbuild we want to make sure that
    # the ctest from this rblxbuild is used for testing
    # and not the ctest from the rblxbuild building and testing
    # rblxbuild.
    if(RBLXBuild_TEST_EXTERNAL_RBLXBUILD)
      set(RBLXBUILD_CTEST_COMMAND "${RBLXBuild_TEST_EXTERNAL_RBLXBUILD}/ctest")
      set(RBLXBUILD_RBLXBUILD_COMMAND "${RBLXBuild_TEST_EXTERNAL_RBLXBUILD}/rblxbuild")
      set(RBLXBUILD_CPACK_COMMAND "${RBLXBuild_TEST_EXTERNAL_RBLXBUILD}/cpack")
      foreach(exe IN ITEMS rblxbuild ctest cpack)
        add_executable(${exe} IMPORTED)
        set_property(TARGET ${exe} PROPERTY IMPORTED_LOCATION ${RBLXBuild_TEST_EXTERNAL_RBLXBUILD}/${exe})
      endforeach()
    else()
      set(RBLXBUILD_CTEST_COMMAND "${RBLXBuild_BIN_DIR}/ctest")
      set(RBLXBUILD_RBLXBUILD_COMMAND "${RBLXBuild_BIN_DIR}/rblxbuild")
      set(RBLXBUILD_CPACK_COMMAND "${RBLXBuild_BIN_DIR}/cpack")
    endif()
  endif()

  # configure some files for testing
  configure_file(Tests/.NoDartCoverage Tests/.NoDartCoverage)
  configure_file(CTestCustom.rblxbuild.in CTestCustom.rblxbuild @ONLY)
endmacro()

option(RBLXBuild_RUN_CLANG_TIDY "Run clang-tidy with the compiler." OFF)
if(RBLXBuild_RUN_CLANG_TIDY)
  if(RBLXBuild_SOURCE_DIR STREQUAL RBLXBuild_BINARY_DIR)
    message(FATAL_ERROR "RBLXBuild_RUN_CLANG_TIDY requires an out-of-source build!")
  endif()
  find_program(CLANG_TIDY_COMMAND NAMES clang-tidy)
  if(NOT CLANG_TIDY_COMMAND)
    message(FATAL_ERROR "RBLXBuild_RUN_CLANG_TIDY is ON but clang-tidy is not found!")
  endif()
  set(RBLXBUILD_CXX_CLANG_TIDY "${CLANG_TIDY_COMMAND}")

  option(RBLXBuild_USE_CLANG_TIDY_MODULE "Use RBLXBuild's clang-tidy module." OFF)
  if(RBLXBuild_USE_CLANG_TIDY_MODULE)
    find_library(RBLXBuild_CLANG_TIDY_MODULE NAMES rblxbuild-clang-tidy-module DOC "Location of the clang-tidy module")
    if(NOT RBLXBuild_CLANG_TIDY_MODULE)
      message(FATAL_ERROR "RBLXBuild_USE_CLANG_TIDY_MODULE is ON but rblxbuild-clang-tidy-module is not found!")
    endif()
    list(APPEND RBLXBUILD_CXX_CLANG_TIDY "--load=${RBLXBuild_CLANG_TIDY_MODULE}")
    set_property(DIRECTORY APPEND PROPERTY RBLXBUILD_CONFIGURE_DEPENDS "${RBLXBuild_CLANG_TIDY_MODULE}")
  endif()

  set(RBLXBuild_CLANG_TIDY_EXPORT_FIXES_DIR "" CACHE PATH "Directory to put clang-tidy fix files in.")
  mark_as_advanced(RBLXBuild_CLANG_TIDY_EXPORT_FIXES_DIR)
  if(RBLXBuild_CLANG_TIDY_EXPORT_FIXES_DIR)
    if(NOT IS_ABSOLUTE "${RBLXBuild_CLANG_TIDY_EXPORT_FIXES_DIR}")
      message(FATAL_ERROR "RBLXBuild_CLANG_TIDY_EXPORT_FIXES_DIR must be an absolute path!")
    endif()
    set(RBLXBUILD_CXX_CLANG_TIDY_EXPORT_FIXES_DIR "${RBLXBuild_CLANG_TIDY_EXPORT_FIXES_DIR}")
  endif()

  # Create a preprocessor definition that depends on .clang-tidy content so
  # the compile command will change when .clang-tidy changes.  This ensures
  # that a subsequent build re-runs clang-tidy on all sources even if they
  # do not otherwise need to be recompiled.  Nothing actually uses this
  # definition.  We add it to targets on which we run clang-tidy just to
  # get the build dependency on the .clang-tidy file.
  file(SHA1 ${RBLXBUILD_CURRENT_SOURCE_DIR}/.clang-tidy clang_tidy_sha1)
  set(CLANG_TIDY_DEFINITIONS "CLANG_TIDY_SHA1=${clang_tidy_sha1}")
  unset(clang_tidy_sha1)
  if(RBLXBuild_USE_CLANG_TIDY_MODULE)
    file(SHA1 "${RBLXBuild_CLANG_TIDY_MODULE}" clang_tidy_module_sha1)
    list(APPEND CLANG_TIDY_DEFINITIONS "CLANG_TIDY_MODULE_SHA1=${clang_tidy_module_sha1}")
    unset(clang_tidy_module_sha1)
  endif()

  configure_file(.clang-tidy .clang-tidy COPYONLY)
endif()

# a macro that only sets the FOLDER target property if it's
# "appropriate"

macro(RBLXBUILD_SET_TARGET_FOLDER tgt folder)
  if(RBLXBUILD_USE_FOLDERS)
    set_property(GLOBAL PROPERTY USE_FOLDERS ON)
    if(MSVC AND TARGET ${tgt})
      set_property(TARGET "${tgt}" PROPERTY FOLDER "${folder}")
    endif()
  else()
    set_property(GLOBAL PROPERTY USE_FOLDERS OFF)
  endif()
endmacro()

    set(RBLXBUILD_INSTALL_RPATH_USE_LINK_PATH ON CACHE BOOL "Install with RPATH set to find custom-built libraries.")
    set(RBLXBUILD_BUILD_WITH_INSTALL_RPATH ON CACHE BOOL "Build with RPATH set to match install-tree RPATH.")
    mark_as_advanced(RBLXBUILD_INSTALL_RPATH_USE_LINK_PATH RBLXBUILD_BUILD_WITH_INSTALL_RPATH)
  endif()

  # add the uninstall support
  configure_file(rblxbuild_uninstall.rblxbuild.in rblxbuild_uninstall.rblxbuild @ONLY)
  add_custom_target(uninstall
    "${RBLXBUILD_COMMAND}" -P "${RBLXBUILD_CURRENT_BINARY_DIR}/rblxbuild_uninstall.rblxbuild")

  include(RBLXBuildCPack.rblxbuild)

endif()
