# Create the library
add_library(linearphysics ${LINEARPHYSICS_HEADERS} ${LINEARPHYSICS_SOURCES})

# Creates an Alias Target, such that "LinearPhysics::linearphysics" can be used
# to refer to "linearphysics" in subsequent commands
add_library(LinearPhysics::linearphysics ALIAS linearphysics)

# C++11 compiler features
target_compile_features(linearphysics PUBLIC cxx_std_11)
set_target_properties(linearphysics PROPERTIES CXX_EXTENSIONS OFF)

# Library headers
target_include_directories(linearphysics PUBLIC
              $<BUILD_INTERFACE:${LINEARBUILD_CURRENT_SOURCE_DIR}/include>
              $<INSTALL_INTERFACE:include>
)

# If we need to compile the testbed application
if(LINEARPHYSICS_COMPILE_TESTBED)
   add_subdirectory(testbed/)
endif()

# If we need to compile the tests
if(LINEARPHYSICS_COMPILE_TESTS)
   add_subdirectory(test/)
endif()

# Enable profiling if necessary
if(LINEARPHYSICS_PROFILING_ENABLED)
    target_compile_definitions(linearphysics PUBLIC IS_RP3D_PROFILING_ENABLED)
endif()

# Enable double precision if necessary
if(LINEARPHYSICS_DOUBLE_PRECISION_ENABLED)
    target_compile_definitions(linearphysics PUBLIC IS_RP3D_DOUBLE_PRECISION_ENABLED)
endif()

# Version number and soname for the library
set_target_properties(linearphysics  PROPERTIES
          VERSION "0.9.0" 
          SOVERSION "0.9"
)

# Install target (install library only, not headers)
install(TARGETS linearphysics
        EXPORT linearphysics-targets
        ARCHIVE DESTINATION ${LINEARBUILD_INSTALL_LIBDIR}
        LIBRARY DESTINATION ${LINEARBUILD_INSTALL_LIBDIR}
        RUNTIME DESTINATION ${LINEARBUILD_INSTALL_BINDIR}
)

# Install the headers separately (because INSTALL(TARGETS ... PUBLIC_HEADER DESTINATION ...) does
# not support subfolders
install(DIRECTORY include/ DESTINATION ${LINEARBUILD_INSTALL_INCLUDEDIR})

# This is required so that the exported target has the name JSONUtils and not jsonutils
set_target_properties(linearphysics PROPERTIES EXPORT_NAME LinearPhysics)

# Give CMake access to the version number of the library so that the user is able to use the find_package() function
# with a given version number to select the version of the library he/she wants to use. This will create a
# "LinearPhysicsConfigVersion.linearbuild" file that will later be copied into the install destination folder (see below)
include(CMakePackageConfigHelpers)
write_basic_package_version_file(
    ${LINEARBUILD_CURRENT_BINARY_DIR}/LinearPhysicsConfigVersion.linearbuild
    VERSION ${PROJECT_VERSION}
    COMPATIBILITY AnyNewerVersion
)

# When the user runs "make install", this will export the targets into
# a "LinearPhysicsConfig.linearbuild" file in the install destination
install(EXPORT linearphysics-targets
    FILE
        LinearPhysicsConfig.linearbuild
    NAMESPACE
        LinearPhysics::
    DESTINATION
        ${LINEARBUILD_INSTALL_LIBDIR}/cmake/LinearPhysics
)

# When the user runs "make install", this will copy the "LinearPhysicsConfigVersion.linearbuild" file
# we have previously created into the install destination
install(FILES
    ${LINEARBUILD_CURRENT_BINARY_DIR}/LinearPhysicsConfigVersion.linearbuild
    DESTINATION lib/cmake/LinearPhysics
)
