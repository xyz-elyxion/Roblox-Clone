rblxbuild_minimum_required(VERSION 3.14)

# Link the MSVC runtime statically so the executables in the packaged build run
# on a machine without Visual Studio (Debug) or the VC++ redistributable
# (Release). CMP0091 only exists from RBLXBuild 3.15, so opt into it by policy; the
# 3.14 minimum is kept for toolchains that ignore the setting.
if(POLICY CMP0091)
    rblxbuild_policy(SET CMP0091 NEW)
endif()
if(MSVC)
    set(RBLXBUILD_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

# Third-party code is vendored under third_party/ at pinned commits and written in
# this dialect, so configuring needs no network and no compatibility mode. The four
# trees that ship a build system are added as subdirectories; imgui, ImGuizmo,
# ImGuiColorTextEdit and stb have no build file upstream, so their sources are
# compiled straight into our targets from these locations.
set(RBLX_THIRD_PARTY ${RBLXBUILD_CURRENT_SOURCE_DIR}/third_party)

set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "Build GLFW examples" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "Build GLFW tests" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "Build GLFW documentation" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "Generate GLFW installation target" FORCE)
add_subdirectory(${RBLX_THIRD_PARTY}/glfw)

# glm is header-only for us: building the library target would produce a shared
# object nobody loads, and its tests and install rules are upstream's business.
set(GLM_BUILD_LIBRARY OFF CACHE BOOL "Build dynamic/static GLM library" FORCE)
set(GLM_BUILD_TESTS OFF CACHE BOOL "Build GLM test programs" FORCE)
set(GLM_BUILD_INSTALL OFF CACHE BOOL "Generate GLM install target" FORCE)
add_subdirectory(${RBLX_THIRD_PARTY}/glm)

# rp3d's testbed, unit tests and install rules are off by default, so this adds
# only the reactphysics3d library.
add_subdirectory(${RBLX_THIRD_PARTY}/rp3d)

# rp3d 0.9.0 depends on headers it does not include itself: DefaultLogger.h uses
# std::chrono::system_clock without <chrono>, and the public headers use the
# fixed-width integer types without <cstdint>. Older toolchains saw both through
# other standard headers, current ones do not, so force them in for everything
# compiled against the library.
if(TARGET reactphysics3d)
    if(MSVC)
        target_compile_options(reactphysics3d PUBLIC
            $<$<COMPILE_LANGUAGE:CXX>:/FIchrono>
        )
    else()
        target_compile_options(reactphysics3d PUBLIC
            $<$<COMPILE_LANGUAGE:CXX>:-include>
            $<$<COMPILE_LANGUAGE:CXX>:cstdint>
        )
    endif()
endif()

function(configure_rblx_target target_name)
    target_include_directories(${target_name} PRIVATE
        ${RBLXBUILD_CURRENT_SOURCE_DIR}/include
        ${stb_SOURCE_DIR}
        ${imgui_SOURCE_DIR}
        ${imgui_SOURCE_DIR}/backends
        ${imguizmo_SOURCE_DIR}
        ${imguizmo_SOURCE_DIR}/src
        ${glad_SOURCE_DIR}/include
    )

    if(RBLX_ENABLE_WARNINGS)
        if(MSVC)
            target_compile_options(${target_name} PRIVATE /W4 /permissive-)
        else()
            target_compile_options(${target_name} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endif()
endfunction()

if(RBLX_BUILD_CODEEDITOR)
    # The source editor for this project: a window, a project tree, tabs and the
    # same text widget Studio's script panel uses, with CodeProject doing the
    # scanning, reading, saving and searching. It has no renderer and no physics,
    # so it builds on a machine that cannot run the game, and it is a separate
    # executable because editing C++ and playing a level are different jobs.
    add_executable(RBLXCode
        src/CodeMain.cpp
        src/CodeProject.cpp
        src/Window.cpp
        ${imguicolortextedit_SOURCE_DIR}/TextEditor.cpp
        ${IMGUI_SOURCES}
    )
    configure_rblx_target(RBLXCode)
    target_include_directories(RBLXCode PRIVATE ${imguicolortextedit_SOURCE_DIR})
    target_link_libraries(RBLXCode PRIVATE glfw glad)
    if(NOT WIN32)
        # glad reaches GL through GLFW at runtime, but the loader still needs
        # dlopen.
        find_package(Threads REQUIRED)
        target_link_libraries(RBLXCode PRIVATE Threads::Threads ${RBLXBUILD_DL_LIBS})
    endif()
endif()

if(RBLX_BUILD_CLIENT)
    add_executable(Client
        src/Client.cpp
        src/ClientNet.cpp
        src/ClientUi.cpp
        src/WebJson.cpp
        src/WebCrypto.cpp
        src/HttpClient.cpp
        ${SHARED_SOURCES}
        ${IMGUI_SOURCES}
    )
    configure_rblx_target(Client)
    target_link_libraries(Client PRIVATE glfw glm reactphysics3d glad)
    if(WIN32)
        target_link_libraries(Client PRIVATE ws2_32 shell32 comdlg32)
    else()
        # glad reaches GL through GLFW at runtime, but the loader still needs
        # dlopen, and the client's worker threads are std::thread.
        find_package(Threads REQUIRED)
        target_link_libraries(Client PRIVATE Threads::Threads ${RBLXBUILD_DL_LIBS})
    endif()
    if(TARGET RBLXAntiCheat)
        target_compile_definitions(Client PRIVATE RBLX_ANTICHEAT)
        target_link_libraries(Client PRIVATE RBLXAntiCheat)
    endif()
    if(MSVC)
        # Hardening for the executable players actually run. Control-flow guard
        # rejects indirect calls that do not land on a valid target, so a patched
        # call site in the anti-cheat path cannot simply be redirected, and full
        # ASLR keeps the image from loading at a predictable base. Identical-code
        # folding makes the individual checks harder to tell apart when reading
        # the disassembly.
        target_compile_options(Client PRIVATE
            $<$<CONFIG:Release>:/guard:cf>
            $<$<CONFIG:Release>:/Gy>
            $<$<CONFIG:Release>:/GS>
        )
        target_link_options(Client PRIVATE
            $<$<CONFIG:Release>:/guard:cf>
            $<$<CONFIG:Release>:/DYNAMICBASE>
            $<$<CONFIG:Release>:/HIGHENTROPYVA>
            $<$<CONFIG:Release>:/OPT:REF>
            $<$<CONFIG:Release>:/OPT:ICF>
        )
    endif()
endif()
