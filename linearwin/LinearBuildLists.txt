#--------------------------------------------------------------------
# Apply Microsoft C runtime library option
# This is here because it also applies to tests and examples
#--------------------------------------------------------------------
if (MSVC)
    if (MSVC90)
        # Workaround for VS 2008 not shipping with the DirectX 9 SDK
        include(CheckIncludeFile)
        check_include_file(dinput.h DINPUT_H_FOUND)
        if (NOT DINPUT_H_FOUND)
            message(FATAL_ERROR "DirectX 9 headers not found; install DirectX 9 SDK")
        endif()
        # Workaround for VS 2008 not shipping with stdint.h
        list(APPEND linearwin_INCLUDE_DIRS "${LINEARWIN_SOURCE_DIR}/deps/vs2008")
    endif()
endif()

if (MSVC AND NOT USE_MSVC_RUNTIME_LIBRARY_DLL)
    if (LINEARBUILD_VERSION VERSION_LESS 3.15)
        foreach (flag LINEARBUILD_C_FLAGS
                      LINEARBUILD_C_FLAGS_DEBUG
                      LINEARBUILD_C_FLAGS_RELEASE
                      LINEARBUILD_C_FLAGS_MINSIZEREL
                      LINEARBUILD_C_FLAGS_RELWITHDEBINFO)

            if (flag MATCHES "/MD")
                string(REGEX REPLACE "/MD" "/MT" ${flag} "${${flag}}")
            endif()
            if (flag MATCHES "/MDd")
                string(REGEX REPLACE "/MDd" "/MTd" ${flag} "${${flag}}")
            endif()

        endforeach()
    else()
        set(LINEARBUILD_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
endif()

if (MINGW)
    # Workaround for legacy MinGW not providing XInput and DirectInput
    include(CheckIncludeFile)

    check_include_file(dinput.h DINPUT_H_FOUND)
    check_include_file(xinput.h XINPUT_H_FOUND)
    if (NOT DINPUT_H_FOUND OR NOT XINPUT_H_FOUND)
        list(APPEND linearwin_INCLUDE_DIRS "${LINEARWIN_SOURCE_DIR}/deps/mingw")
    endif()

    # Enable link-time exploit mitigation features enabled by default on MSVC
    include(CheckCCompilerFlag)

    # Compatibility with data execution prevention (DEP)
    set(LINEARBUILD_REQUIRED_FLAGS "-Wl,--nxcompat")
    check_c_compiler_flag("" _LINEARWIN_HAS_DEP)
    if (_LINEARWIN_HAS_DEP)
        set(LINEARBUILD_SHARED_LINKER_FLAGS "-Wl,--nxcompat ${LINEARBUILD_SHARED_LINKER_FLAGS}")
    endif()

    # Compatibility with address space layout randomization (ASLR)
    set(LINEARBUILD_REQUIRED_FLAGS "-Wl,--dynamicbase")
    check_c_compiler_flag("" _LINEARWIN_HAS_ASLR)
    if (_LINEARWIN_HAS_ASLR)
        set(LINEARBUILD_SHARED_LINKER_FLAGS "-Wl,--dynamicbase ${LINEARBUILD_SHARED_LINKER_FLAGS}")
    endif()

    # Compatibility with 64-bit address space layout randomization (ASLR)
    set(LINEARBUILD_REQUIRED_FLAGS "-Wl,--high-entropy-va")
    check_c_compiler_flag("" _LINEARWIN_HAS_64ASLR)
    if (_LINEARWIN_HAS_64ASLR)
        set(LINEARBUILD_SHARED_LINKER_FLAGS "-Wl,--high-entropy-va ${LINEARBUILD_SHARED_LINKER_FLAGS}")
    endif()

    # Clear flags again to avoid breaking later tests
    set(LINEARBUILD_REQUIRED_FLAGS)
endif()

if (LINEARWIN_USE_WAYLAND)
    set(_LINEARWIN_WAYLAND 1)
    message(STATUS "Using Wayland for window creation")

    find_library(RT_LIBRARY rt)
    mark_as_advanced(RT_LIBRARY)
    if (RT_LIBRARY)
        list(APPEND linearwin_LIBRARIES "${RT_LIBRARY}")
        list(APPEND linearwin_PKG_LIBS "-lrt")
    endif()

    find_library(MATH_LIBRARY m)
    mark_as_advanced(MATH_LIBRARY)
    if (MATH_LIBRARY)
        list(APPEND linearwin_LIBRARIES "${MATH_LIBRARY}")
        list(APPEND linearwin_PKG_LIBS "-lm")
    endif()

    if (LINEARBUILD_DL_LIBS)
        list(APPEND linearwin_LIBRARIES "${LINEARBUILD_DL_LIBS}")
        list(APPEND linearwin_PKG_LIBS "-l${LINEARBUILD_DL_LIBS}")
    endif()
endif()

#--------------------------------------------------------------------
# Use X11 for window creation
#--------------------------------------------------------------------
if (_LINEARWIN_X11)

    find_package(X11 REQUIRED)

    list(APPEND linearwin_PKG_DEPS "x11")

    # Set up library and include paths
    list(APPEND linearwin_INCLUDE_DIRS "${X11_X11_INCLUDE_PATH}")
    list(APPEND linearwin_LIBRARIES "${X11_X11_LIB}" "${LINEARBUILD_THREAD_LIBS_INIT}")

    # Check for XRandR (modern resolution switching and gamma control)
    if (NOT X11_Xrandr_INCLUDE_PATH)
        message(FATAL_ERROR "RandR headers not found; install libxrandr development package")
    endif()

    # Check for Xinerama (legacy multi-monitor support)
    if (NOT X11_Xinerama_INCLUDE_PATH)
        message(FATAL_ERROR "Xinerama headers not found; install libxinerama development package")
    endif()

    # Check for Xkb (X keyboard extension)
    if (NOT X11_Xkb_INCLUDE_PATH)
        message(FATAL_ERROR "XKB headers not found; install X11 development package")
    endif()

    # Check for Xcursor (cursor creation from RGBA images)
    if (NOT X11_Xcursor_INCLUDE_PATH)
        message(FATAL_ERROR "Xcursor headers not found; install libxcursor development package")
    endif()

    # Check for XInput (modern HID input)
    if (NOT X11_Xi_INCLUDE_PATH)
        message(FATAL_ERROR "XInput headers not found; install libxi development package")
    endif()

    list(APPEND linearwin_INCLUDE_DIRS "${X11_Xrandr_INCLUDE_PATH}"
                                  "${X11_Xinerama_INCLUDE_PATH}"
                                  "${X11_Xkb_INCLUDE_PATH}"
                                  "${X11_Xcursor_INCLUDE_PATH}"
                                  "${X11_Xi_INCLUDE_PATH}")
endif()

    list(APPEND linearwin_PKG_DEPS "wayland-client")

    list(APPEND linearwin_INCLUDE_DIRS "${Wayland_INCLUDE_DIRS}")
    list(APPEND linearwin_LIBRARIES "${Wayland_LIBRARIES}" "${LINEARBUILD_THREAD_LIBS_INIT}")

    find_package(XKBCommon REQUIRED)
    list(APPEND linearwin_INCLUDE_DIRS "${XKBCOMMON_INCLUDE_DIRS}")

    include(CheckIncludeFiles)
    include(CheckFunctionExists)

    if (NOT LINEARBUILD_SYSTEM_NAME STREQUAL "Linux")
        find_package(EpollShim)

            list(APPEND linearwin_LIBRARIES "${EPOLLSHIM_LIBRARIES}")
        endif()
    endif()
endif()

            DESTINATION "${LINEARBUILD_INSTALL_LIBDIR}/pkgconfig")
