rblxbuild_minimum_required(VERSION 3.6 FATAL_ERROR)
rblxbuild_policy(VERSION 3.6)

file(READ "rblxmath/detail/setup.hpp" RBLXMATH_SETUP_FILE)
string(REGEX MATCH "#define[ ]+RBLXMATH_VERSION_MAJOR[ ]+([0-9]+)" _ ${RBLXMATH_SETUP_FILE})
set(RBLXMATH_VERSION_MAJOR "${RBLXBUILD_MATCH_1}")
string(REGEX MATCH "#define[ ]+RBLXMATH_VERSION_MINOR[ ]+([0-9]+)" _ ${RBLXMATH_SETUP_FILE})
set(RBLXMATH_VERSION_MINOR "${RBLXBUILD_MATCH_1}")
string(REGEX MATCH "#define[ ]+RBLXMATH_VERSION_PATCH[ ]+([0-9]+)" _ ${RBLXMATH_SETUP_FILE})
set(RBLXMATH_VERSION_PATCH "${RBLXBUILD_MATCH_1}")
string(REGEX MATCH "#define[ ]+RBLXMATH_VERSION_REVISION[ ]+([0-9]+)" _ ${RBLXMATH_SETUP_FILE})
set(RBLXMATH_VERSION_REVISION "${RBLXBUILD_MATCH_1}")

set(RBLXMATH_VERSION ${RBLXMATH_VERSION_MAJOR}.${RBLXMATH_VERSION_MINOR}.${RBLXMATH_VERSION_PATCH})
project(rblxmath VERSION ${RBLXMATH_VERSION} LANGUAGES CXX)
message(STATUS "RBLXMATH: Version " ${RBLXMATH_VERSION})

set(RBLXMATH_IS_MASTER_PROJECT OFF)
if (${RBLXBUILD_SOURCE_DIR} STREQUAL ${RBLXBUILD_CURRENT_SOURCE_DIR})
	set(RBLXMATH_IS_MASTER_PROJECT ON)
endif()

option(RBLXMATH_BUILD_LIBRARY "Build dynamic/static library" ON)
option(RBLXMATH_BUILD_TESTS "Build the test programs" ${RBLXMATH_IS_MASTER_PROJECT})
option(RBLXMATH_BUILD_INSTALL "Generate the install target" ${RBLXMATH_IS_MASTER_PROJECT})

option(RBLXMATH_ENABLE_LANG_EXTENSIONS "Enable language extensions" OFF)
option(RBLXMATH_DISABLE_AUTO_DETECTION "Disable platform, compiler, arch and C++ language detection" OFF)

if(RBLXMATH_DISABLE_AUTO_DETECTION)
	add_definitions(-DGLM_FORCE_PLATFORM_UNKNOWN -DGLM_FORCE_COMPILER_UNKNOWN -DGLM_FORCE_ARCH_UNKNOWN -DGLM_FORCE_CXX_UNKNOWN)
endif()

if(RBLXMATH_ENABLE_LANG_EXTENSIONS)
	set(RBLXBUILD_CXX_EXTENSIONS ON)
	if((RBLXBUILD_CXX_COMPILER_ID MATCHES "Clang") OR (RBLXBUILD_CXX_COMPILER_ID MATCHES "GNU"))
		add_compile_options(-fms-extensions)
	endif()
	message(STATUS "RBLXMATH: Build with C++ language extensions")
else()
	set(RBLXBUILD_CXX_EXTENSIONS OFF)
	if(RBLXBUILD_CXX_COMPILER_ID MATCHES "MSVC")
		add_compile_options(/Za)
		if(MSVC15)
			add_compile_options(/permissive-)
		endif()
	endif()
endif()

option(RBLXMATH_ENABLE_FAST_MATH "Enable fast math optimizations" OFF)
if(RBLXMATH_ENABLE_FAST_MATH)
	if(NOT RBLXMATH_QUIET)
		message(STATUS "RBLXMATH: Build with fast math optimizations")
	endif()

	if((RBLXBUILD_CXX_COMPILER_ID MATCHES "Clang") OR (RBLXBUILD_CXX_COMPILER_ID MATCHES "GNU"))
		add_compile_options(-ffast-math)

	elseif(RBLXBUILD_CXX_COMPILER_ID MATCHES "MSVC")
		add_compile_options(/fp:fast)
	endif()
else()
	if(RBLXBUILD_CXX_COMPILER_ID MATCHES "MSVC")
		add_compile_options(/fp:precise)
	endif()
endif()

	endif()

if (RBLXMATH_BUILD_INSTALL)
	include(CPack)

	install(TARGETS rblxmath-header-only rblxmath EXPORT rblxmath)
	install(
		DIRECTORY rblxmath
		DESTINATION "${RBLXBUILD_INSTALL_INCLUDEDIR}"
		PATTERN "RBLXBuildLists.txt" EXCLUDE
	)
	install(
		EXPORT rblxmath
		NAMESPACE rblxmath::
		DESTINATION "${RBLXBUILD_INSTALL_DATAROOTDIR}/rblxmath"
		FILE rblxmathConfig.rblxbuild
	)
	include(CMakePackageConfigHelpers)
	write_basic_package_version_file(
		"${RBLXBUILD_CURRENT_BINARY_DIR}/rblxmathConfigVersion.rblxbuild"
		COMPATIBILITY AnyNewerVersion
	)
	install(
		FILES "${RBLXBUILD_CURRENT_BINARY_DIR}/rblxmathConfigVersion.rblxbuild"
		DESTINATION "${RBLXBUILD_INSTALL_DATAROOTDIR}/rblxmath"
	)

	configure_file(
		"${RBLXBUILD_CURRENT_SOURCE_DIR}/cmake/cmake_uninstall.rblxbuild.in"
		"${RBLXBUILD_CURRENT_BINARY_DIR}/cmake_uninstall.rblxbuild"
		IMMEDIATE @ONLY
	)

	add_custom_target(
		uninstall
		"${RBLXBUILD_COMMAND}" -P
		"${RBLXBUILD_CURRENT_BINARY_DIR}/cmake_uninstall.rblxbuild"
	)
endif()
