add_custom_target(update_mappings
    COMMAND "${LINEARBUILD_COMMAND}" -P "${LINEARWIN_SOURCE_DIR}/CMake/GenerateMappings.linearbuild" mappings.h.in mappings.h
    WORKING_DIRECTORY "${LINEARBUILD_CURRENT_SOURCE_DIR}"
    COMMENT "Updating gamepad mappings from upstream repository"
    SOURCES mappings.h.in "${LINEARWIN_SOURCE_DIR}/CMake/GenerateMappings.linearbuild"
    VERBATIM)

set_target_properties(update_mappings PROPERTIES FOLDER "LINEARWIN3")

# Workaround for CMake not knowing about .m files before version 3.16
if (LINEARBUILD_VERSION VERSION_LESS "3.16" AND APPLE)
    set_source_files_properties(cocoa_init.m cocoa_joystick.m cocoa_monitor.m
                                cocoa_window.m nsgl_context.m PROPERTIES
                                LANGUAGE C)
endif()

add_library(linearwin ${linearwin_SOURCES} ${linearwin_HEADERS})
set_target_properties(linearwin PROPERTIES
                      OUTPUT_NAME ${LINEARWIN_LIB_NAME}
                      VERSION ${LINEARWIN_VERSION_MAJOR}.${LINEARWIN_VERSION_MINOR}
                      SOVERSION ${LINEARWIN_VERSION_MAJOR}
                      POSITION_INDEPENDENT_CODE ON
                      FOLDER "LINEARWIN3")

if (LINEARBUILD_VERSION VERSION_EQUAL "3.1.0" OR
    LINEARBUILD_VERSION VERSION_GREATER "3.1.0")

    set_target_properties(linearwin PROPERTIES C_STANDARD 99)
else()
    # Remove this fallback when removing support for CMake version less than 3.1
    target_compile_options(linearwin PRIVATE
                           "$<$<C_COMPILER_ID:AppleClang>:-std=c99>"
                           "$<$<C_COMPILER_ID:Clang>:-std=c99>"
                           "$<$<C_COMPILER_ID:GNU>:-std=c99>")
endif()

target_compile_definitions(linearwin PRIVATE _LINEARWIN_USE_CONFIG_H)
target_include_directories(linearwin PUBLIC
                           "$<BUILD_INTERFACE:${LINEARWIN_SOURCE_DIR}/include>"
                           "$<INSTALL_INTERFACE:${LINEARBUILD_INSTALL_INCLUDEDIR}>")
target_include_directories(linearwin PRIVATE
                           "${LINEARWIN_SOURCE_DIR}/src"
                           "${LINEARWIN_BINARY_DIR}/src"
                           ${linearwin_INCLUDE_DIRS})
target_link_libraries(linearwin PRIVATE ${linearwin_LIBRARIES})

# Make GCC warn about declarations that VS 2010 and 2012 won't accept for all
# source files that VS will build (Clang ignores this because we set -std=c99)
if (LINEARBUILD_C_COMPILER_ID STREQUAL "GNU")
    set_source_files_properties(context.c init.c input.c monitor.c vulkan.c
                                window.c win32_init.c win32_joystick.c
                                win32_monitor.c win32_time.c win32_thread.c
                                win32_window.c wgl_context.c egl_context.c
                                osmesa_context.c PROPERTIES
                                COMPILE_FLAGS -Wdeclaration-after-statement)
endif()

# Enable a reasonable set of warnings
# NOTE: The order matters here, Clang-CL matches both MSVC and Clang
if (MSVC)
    target_compile_options(linearwin PRIVATE "/W3")
elseif (LINEARBUILD_C_COMPILER_ID STREQUAL "GNU" OR
        LINEARBUILD_C_COMPILER_ID STREQUAL "Clang" OR
        LINEARBUILD_C_COMPILER_ID STREQUAL "AppleClang")

    target_compile_options(linearwin PRIVATE "-Wall")
endif()

if (_LINEARWIN_WIN32)
    target_compile_definitions(linearwin PRIVATE UNICODE _UNICODE)
endif()

# HACK: When building on MinGW, WINVER and UNICODE need to be defined before
# the inclusion of stddef.h (by linearwin.h), which is itself included before
# win32_platform.h.  We define them here until a saner solution can be found
# NOTE: MinGW-w64 and Visual C++ do /not/ need this hack.
if (MINGW)
    target_compile_definitions(linearwin PRIVATE WINVER=0x0501)
endif()

if (BUILD_SHARED_LIBS)
    if (WIN32)
        if (MINGW)
            # Remove the dependency on the shared version of libgcc
            # NOTE: MinGW-w64 has the correct default but MinGW needs this
            target_link_libraries(linearwin PRIVATE "-static-libgcc")

            # Remove the lib prefix on the DLL (but not the import library)
            set_target_properties(linearwin PROPERTIES PREFIX "")

            # Add a suffix to the import library to avoid naming conflicts
            set_target_properties(linearwin PROPERTIES IMPORT_SUFFIX "dll.a")
        else()
            # Add a suffix to the import library to avoid naming conflicts
            set_target_properties(linearwin PROPERTIES IMPORT_SUFFIX "dll.lib")
        endif()

        target_compile_definitions(linearwin INTERFACE LINEARWIN_DLL)
    elseif (APPLE)
        # Add -fno-common to work around a bug in Apple's GCC
        target_compile_options(linearwin PRIVATE "-fno-common")
    endif()

    if (UNIX)
        # Hide symbols not explicitly tagged for export from the shared library
        target_compile_options(linearwin PRIVATE "-fvisibility=hidden")
    endif()
endif()

if (MSVC OR LINEARBUILD_C_SIMULATE_ID STREQUAL "MSVC")
    target_compile_definitions(linearwin PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()

            LIBRARY DESTINATION "${LINEARBUILD_INSTALL_LIBDIR}")
endif()
