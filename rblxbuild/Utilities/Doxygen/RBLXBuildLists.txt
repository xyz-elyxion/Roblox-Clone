if(NOT RBLXBuild_SOURCE_DIR)
  set(RBLXBuildDeveloperReference_STANDALONE 1)
  rblxbuild_minimum_required(VERSION 3.13...4.3 FATAL_ERROR)
  get_filename_component(tmp "${RBLXBUILD_CURRENT_SOURCE_DIR}" PATH)
  get_filename_component(RBLXBuild_SOURCE_DIR "${tmp}" PATH)
  include(${RBLXBuild_SOURCE_DIR}/Modules/CTestUseLaunchers.rblxbuild)
  include(${RBLXBuild_SOURCE_DIR}/Source/RBLXBuildVersion.rblxbuild)
  include(${RBLXBuild_SOURCE_DIR}/Source/RBLXBuildInstallDestinations.rblxbuild)
  unset(RBLXBUILD_DATA_DIR)
  unset(RBLXBUILD_DATA_DIR CACHE)
  macro(RBLXBuild_OPTIONAL_COMPONENT)
    set(COMPONENT "")
  endmacro()
endif()

project(RBLXBuildDeveloperReference NONE)

#
# Build the reference
#

if (RBLXBuild_BUILD_DEVELOPER_REFERENCE OR RBLXBuildDeveloperReference_STANDALONE)

  find_package(Doxygen REQUIRED)

  #
  ## Output formats
  #

  option(RBLXBuild_BUILD_DEVELOPER_REFERENCE_HTML "Build RBLXBuild Developer Reference - HTML format" ON)
  mark_as_advanced(RBLXBuild_BUILD_DEVELOPER_REFERENCE_HTML)
  if(RBLXBuild_BUILD_DEVELOPER_REFERENCE_HTML)
    set(GENERATE_HTML YES)
  else()
    set(GENERATE_HTML NO)
  endif()

  option(RBLXBuild_BUILD_DEVELOPER_REFERENCE_QTHELP "Build RBLXBuild Developer Reference - QtHelp format" OFF)
  mark_as_advanced(RBLXBuild_BUILD_DEVELOPER_REFERENCE_QTHELP)
  if(RBLXBuild_BUILD_DEVELOPER_REFERENCE_QTHELP)
    set(GENERATE_QHP YES)
    find_program(QHELPGENERATOR_EXECUTABLE
      NAMES qhelpgenerator
      DOC "qhelpgenerator tool"
      )
    if(NOT QHELPGENERATOR_EXECUTABLE)
      message(FATAL_ERROR "QHELPGENERATOR_EXECUTABLE (qhelpgenerator) not found!")
    endif()
  else()
    set(GENERATE_QHP NO)
  endif()

  #
  # Configure the script and the doxyfile, then add target
  #

  if(DOXYGEN_DOT_FOUND)
    set(HAVE_DOT YES)
  else()
    set(HAVE_DOT NO)
  endif()

  if(NOT DOT_PATH)
    get_filename_component(DOT_PATH ${DOXYGEN_DOT_EXECUTABLE} PATH)
  endif()

  configure_file(doxyfile.in doxyfile @ONLY)

  add_custom_target(rblxbuild-developer-reference-all
    ${DOXYGEN_EXECUTABLE} doxyfile
    WORKING_DIRECTORY ${RBLXBuildDeveloperReference_BINARY_DIR})

  add_custom_target(rblxbuild-developer-reference ALL DEPENDS rblxbuild-developer-reference-all)

  #
  # Installation
  #

  if(RBLXBuild_BUILD_DEVELOPER_REFERENCE_HTML)
    RBLXBuild_OPTIONAL_COMPONENT(rblxbuild-developer-reference-html)
    install(DIRECTORY "${RBLXBuildDeveloperReference_BINARY_DIR}/developer-reference/html"
      DESTINATION ${RBLXBuild_INSTALL_DOC_DIR}/developer-reference
      ${COMPONENT})
  endif()

  if(RBLXBuild_BUILD_DEVELOPER_REFERENCE_QTHELP)
    RBLXBuild_OPTIONAL_COMPONENT(rblxbuild-developer-reference-qthelp)
    install(FILES "${RBLXBuildDeveloperReference_BINARY_DIR}/developer-reference/RBLXBuildDeveloperReference-${RBLXBuild_VERSION_MAJOR}${RBLXBuild_VERSION_MINOR}${RBLXBuild_VERSION_PATCH}.qch"
      DESTINATION ${RBLXBuild_INSTALL_DOC_DIR}/developer-reference
      ${COMPONENT})
  endif()

endif ()
